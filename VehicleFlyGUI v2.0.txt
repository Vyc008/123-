-- Hàm thiết kế viền
local function ApplyMinimalStyle(element)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = element

    local stroke = Instance.new("UIStroke")
    stroke.Color = Theme.Border
    stroke.Thickness = 1
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Parent = element
end

-- Hàm tạo Nút
local function CreateButton(btn, parent, text, size, pos, textSize)
    btn.Parent = parent
    btn.BackgroundColor3 = Theme.Panel
    btn.Size = size
    if pos then btn.Position = pos end
    btn.Font = MainFont
    btn.Text = text
    btn.TextColor3 = Theme.Text
    btn.TextSize = textSize or 12
    
    if btn:IsA("TextButton") then
        btn.AutoButtonColor = true
    end
    
    ApplyMinimalStyle(btn)
end

-- ================= LOGIC BAY & VẬT LÝ =================
local function setupFlyInstances(character)
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    if character.HumanoidRootPart:FindFirstChild(velocityHandlerName) then character.HumanoidRootPart[velocityHandlerName]:Destroy() end
    if character.HumanoidRootPart:FindFirstChild(gyroHandlerName) then character.HumanoidRootPart[gyroHandlerName]:Destroy() end
    
    VelocityHandler = Instance.new("BodyVelocity")
    VelocityHandler.Name = velocityHandlerName
    VelocityHandler.Parent = character.HumanoidRootPart
    VelocityHandler.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    VelocityHandler.Velocity = Vector3.new()
    
    GyroHandler = Instance.new("BodyGyro")
    GyroHandler.Name = gyroHandlerName
    GyroHandler.Parent = character.HumanoidRootPart
    GyroHandler.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    GyroHandler.P = 1000
    GyroHandler.D = 50
    GyroHandler.CFrame = character.HumanoidRootPart.CFrame
end

local function EnableFlight()
    if player.Character then setupFlyInstances(player.Character) end
end

local function DisableFlight()
    if VelocityHandler then VelocityHandler:Destroy() VelocityHandler = nil end
    if GyroHandler then GyroHandler:Destroy() GyroHandler = nil end
end

-- ================= KẾT NỐI SỰ KIỆN =================
local function setupButtonConnections()
    table.insert(activeConnections, IncreaseButton.MouseButton1Click:Connect(function()
        local currentSpeed = tonumber(SpeedTextBox.Text) or 1
        SpeedTextBox.Text = tostring(currentSpeed + 0.1)
    end))
    
    table.insert(activeConnections, DecreaseButton.MouseButton1Click:Connect(function()
        local currentSpeed = tonumber(SpeedTextBox.Text) or 1
        SpeedTextBox.Text = tostring(math.max(0.1, currentSpeed - 0.1))
    end))
    
    local movementButtons = {
        {button = ForwardButton, dir = "forward"}, {button = BackwardButton, dir = "backward"},
        {button = LeftButton, dir = "left"}, {button = RightButton, dir = "right"},
        {button = UpButton, dir = "up"}, {button = DownButton, dir = "down"},
        {button = RotateLeftButton, dir = "rotateLeft"}, {button = RotateRightButton, dir = "rotateRight"}
    }

    for _, config in ipairs(movementButtons) do
        table.insert(activeConnections, config.button.MouseButton1Down:Connect(function() table.insert(movementDirection, config.dir) end))
        table.insert(activeConnections, config.button.MouseButton1Up:Connect(function()
            local index = table.find(movementDirection, config.dir)
            while index do table.remove(movementDirection, index) index = table.find(movementDirection, config.dir) end
        end))
    end
    
    table.insert(activeConnections, AntiLockButton.MouseButton1Click:Connect(function()
        isAntiLockOn = not isAntiLockOn
        AntiLockButton.Text = isAntiLockOn and "A-L: ON" or "A-L: OFF"
        AntiLockButton.BackgroundColor3 = isAntiLockOn and Theme.Active or Theme.Panel
        AntiLockButton.TextColor3 = isAntiLockOn and Theme.Background or Theme.Text

        if isAntiLockOn and isPitchOn then
            isPitchOn = false
            PitchButton.Text = "PITCH: OFF"
            PitchButton.BackgroundColor3 = Theme.Panel
            PitchButton.TextColor3 = Theme.Text
        end
    end))
    
    table.insert(activeConnections, PitchButton.MouseButton1Click:Connect(function()
        if isAntiLockOn then
            isPitchOn = false
        else
            isPitchOn = not isPitchOn
        end
        PitchButton.Text = isPitchOn and "PITCH: ON" or "PITCH: OFF"
        PitchButton.BackgroundColor3 = isPitchOn and Theme.Active or Theme.Panel
        PitchButton.TextColor3 = isPitchOn and Theme.Background or Theme.Text
    end))
end

local function handleInput()
    local character = player.Character
    if not character or not character:FindFirstChild("Humanoid") then return end
    local humanoid = character.Humanoid
    
    if FlyEnabled and (humanoid.SeatPart or flightWithoutFlyEnabled) then
        local rootPart = character:FindFirstChild("HumanoidRootPart")
        if not rootPart or not VelocityHandler then return end
        
        local gyro = rootPart:FindFirstChild(gyroHandlerName)
        local speed = tonumber(SpeedTextBox.Text) or 1
        local velocity = Vector3.new()
        
        local movementCFrame = (isPitchOn and gyro) and gyro.CFrame or (gyro and CFrame.new(gyro.Parent.Position, gyro.Parent.Position + Vector3.new(gyro.CFrame.LookVector.X, 0, gyro.CFrame.LookVector.Z)) or rootPart.CFrame)
        
        for _, direction in ipairs(movementDirection) do
            if direction == "forward" then velocity += movementCFrame.LookVector * forwardVelocity
            elseif direction == "backward" then velocity += -movementCFrame.LookVector * forwardVelocity
            elseif direction == "left" then velocity += movementCFrame.RightVector * -leftVelocity
            elseif direction == "right" then velocity += movementCFrame.RightVector * rightVelocity
            elseif direction == "up" then velocity += Vector3.new(0, upVelocity, 0)
            elseif direction == "down" then velocity += Vector3.new(0, -downVelocity, 0)
            elseif direction == "rotateLeft" and gyro then gyro.CFrame *= CFrame.Angles(0, math.rad(-rotationSpeed), 0)
            elseif direction == "rotateRight" and gyro then gyro.CFrame *= CFrame.Angles(0, math.rad(rotationSpeed), 0)
            end
        end
        
        local thumbstickDirection = Vector2.new(0, 0)
        pcall(function() thumbstickDirection = require(player.PlayerScripts:WaitForChild("PlayerModule"):WaitForChild("ControlModule")):GetMoveVector() end)
        
        if #movementDirection > 0 or thumbstickDirection.Magnitude > 0 then
            local combinedVelocity = velocity + (movementCFrame.RightVector * thumbstickDirection.X * forwardVelocity * 2 + movementCFrame.LookVector * -thumbstickDirection.Z * forwardVelocity * 2)
            if isAntiLockOn and humanoid.SeatPart then
                local sCF = humanoid.SeatPart.CFrame
                local seatVelocity = Vector3.new()
                for _, direction in ipairs(movementDirection) do
                    if direction == "forward" then seatVelocity += sCF.LookVector * forwardVelocity
                    elseif direction == "backward" then seatVelocity += -sCF.LookVector * forwardVelocity
                    elseif direction == "left" then seatVelocity += -sCF.RightVector * leftVelocity
                    elseif direction == "right" then seatVelocity += sCF.RightVector * rightVelocity
                    elseif direction == "up" then seatVelocity += sCF.UpVector * upVelocity
                    elseif direction == "down" then seatVelocity += -sCF.UpVector * downVelocity
                    end
                end
                VelocityHandler.Velocity = seatVelocity * speed
            else
                VelocityHandler.Velocity = combinedVelocity * speed
            end
        else
            VelocityHandler.Velocity = Vector3.new()
        end
        
        if not isAntiLockOn then
            local camera = workspace.CurrentCamera
            if isPitchOn and gyro then gyro.CFrame = camera.CFrame
            elseif gyro then gyro.CFrame = CFrame.new(gyro.Parent.Position, gyro.Parent.Position + (camera.CFrame.LookVector * Vector3.new(1, 0, 1)))
            end
        end
    else
        if VelocityHandler then VelocityHandler.Velocity = Vector3.new() end
    end
end

-- ================= ĐỒNG BỘ TRẠNG THÁI NÚT FLY =================
local function toggleFlyState()
    FlyEnabled = not FlyEnabled
    local humanoid = player.Character and player.Character:FindFirstChild("Humanoid")
    
    local text = FlyEnabled and "FLY: ON" or "FLY: OFF"
    local bgColor = FlyEnabled and Theme.Active or Theme.Panel
    local txtColor = FlyEnabled and Theme.Background or Theme.Text
    
    FlyButton.Text = text
    FlyButton.BackgroundColor3 = bgColor
    FlyButton.TextColor3 = txtColor
    
    FloatingFlyButton.Text = text
    FloatingFlyButton.BackgroundColor3 = bgColor
    FloatingFlyButton.TextColor3 = txtColor
    
    if FlyEnabled then
        if flightWithoutFlyEnabled then EnableFlight()
        elseif humanoid then
            if seatConnection then seatConnection:Disconnect() end
            if unseatConnection then unseatConnection:Disconnect() end
            if humanoid.SeatPart then EnableFlight()
            else seatConnection = humanoid:GetPropertyChangedSignal("SeatPart"):Connect(function() if humanoid.SeatPart then EnableFlight() end end) end
            unseatConnection = humanoid:GetPropertyChangedSignal("SeatPart"):Connect(function() if not humanoid.SeatPart then DisableFlight() end end)
        end
    else
        DisableFlight()
        if seatConnection then seatConnection:Disconnect() end
        if unseatConnection then unseatConnection:Disconnect() end
    end
end

local function initialize()
    initializeGUI()
    setupButtonConnections()
    table.insert(activeConnections, game:GetService("RunService").RenderStepped:Connect(handleInput))
    
    FlyButton.MouseButton1Click:Connect(toggleFlyState)
    FloatingFlyButton.MouseButton1Click:Connect(toggleFlyState)
    
    ToggleFloatingButton.MouseButton1Click:Connect(function()
        isFloatingFlyVisible = not isFloatingFlyVisible
        FloatingFlyButton.Visible = isFloatingFlyVisible
        ToggleFloatingButton.BackgroundColor3 = isFloatingFlyVisible and Theme.Active or Theme.Panel
        ToggleFloatingButton.TextColor3 = isFloatingFlyVisible and Theme.Background or Theme.Text
    end)
    
    FwFButton.MouseButton1Click:Connect(function()
        flightWithoutFlyEnabled = not flightWithoutFlyEnabled
        FwFButton.Text = flightWithoutFlyEnabled and "FwF: ON" or "FwF: OFF"
        FwFButton.BackgroundColor3 = flightWithoutFlyEnabled and Theme.Active or Theme.Panel
        FwFButton.TextColor3 = flightWithoutFlyEnabled and Theme.Background or Theme.Text
        
        if FlyEnabled and not flightWithoutFlyEnabled then
            local humanoid = player.Character and player.Character:FindFirstChild("Humanoid")
            if humanoid and not humanoid.SeatPart then DisableFlight() end
        end
    end)
    
    ControllerButton.MouseButton1Click:Connect(function()
        controllerEnabled = not controllerEnabled
        ControllerGui.Enabled = controllerEnabled
        LeftSideGui.Enabled = controllerEnabled
        ControllerButton.Text = controllerEnabled and "Hide Controller" or "Show Controller"
        ControllerButton.BackgroundColor3 = controllerEnabled and Theme.Active or Theme.Panel
        ControllerButton.TextColor3 = controllerEnabled and Theme.Background or Theme.Text
    end)
    
    MinimizeButton.MouseButton1Click:Connect(function()
        isGuiMinimized = not isGuiMinimized
        if isGuiMinimized then
            MinimizeButton.Text = "+"
            MainFrame:TweenSize(UDim2.new(0, 180, 0, 35), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.3, true)
        else
            MinimizeButton.Text = "–"
            MainFrame:TweenSize(UDim2.new(0, 180, 0, 195), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.3, true)
        end
    end)
end
