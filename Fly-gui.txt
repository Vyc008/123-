local rainbowStrokes = {}
local function addBorder(instance, thickness)
	local stroke = Instance.new("UIStroke")
	stroke.Thickness = thickness or 2
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = instance
	table.insert(rainbowStrokes, stroke)
	return stroke
end

local rainbowConn
rainbowConn = RunService.RenderStepped:Connect(function()
	if not main or not main.Parent then
		if rainbowConn then rainbowConn:Disconnect() end
		return
	end
	local hue = (tick() * 0.4) % 1
	local rainbowColor = Color3.fromHSV(hue, 0.9, 1)
	for _, stroke in ipairs(rainbowStrokes) do
		if stroke and stroke.Parent then
			stroke.Color = rainbowColor
		end
	end
end)

-- Biến hệ thống
local speeds = 1
local nowe = false
local noclip = false
local flyConnection = nil
local noclipConnection = nil
local isUpBtnDown = false
local isDownBtnDown = false
local currentPos = Vector3.zero
local selectedKey = Enum.KeyCode.F

-- HÀM KIỂM TRA MẶT ĐẤT & TRƯỢT VA CHẠY MƯỢT MÀ
local function checkCollision(oldPos, newPos)
	if noclip then return newPos end
	local dir = newPos - oldPos
	if dir.Magnitude < 0.001 then return newPos end

	local char = LocalPlayer.Character
	if not char then return newPos end

	local raycastParams = RaycastParams.new()
	raycastParams.FilterDescendantsInstances = {char}
	raycastParams.FilterType = Enum.RaycastFilterType.Exclude
	raycastParams.IgnoreWater = true

	local result = workspace:Raycast(oldPos, dir, raycastParams)
	if result and result.Instance and result.Instance.CanCollide then
		return result.Position + (result.Normal * 0.2)
	end
	return newPos
end

-- HÀM XỬ LÝ NOCLIP
local function setNoclip(state)
	noclip = state
	if noclip then
		noclipBtn.Text = "NC: ON"
		noclipBtn.BackgroundColor3 = Color3.fromRGB(0, 180, 0)
		if not noclipConnection then
			noclipConnection = RunService.Stepped:Connect(function()
				local char = LocalPlayer.Character
				if char then
					for _, part in ipairs(char:GetDescendants()) do
						if part:IsA("BasePart") then
							part.CanCollide = false
						end
					end
				end
			end)
		end
	else
		noclipBtn.Text = "NC: OFF"
		noclipBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
		if noclipConnection then
			noclipConnection:Disconnect()
			noclipConnection = nil
		end
	end
end

noclipBtn.MouseButton1Click:Connect(function()
	setNoclip(not noclip)
end)

-- HÀM TẮT BAY TỔNG THỂ
local function stopFly()
	nowe = false
	onof.Text = "BAY"
	onof.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
	
	setNoclip(false)
	
	local char = LocalPlayer.Character
	if char then
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then
			hum.PlatformStand = false
		end
	end
	
	if flyConnection then
		flyConnection:Disconnect()
		flyConnection = nil
	end
end

_G.WuWFlyCleanup = stopFly

-- Cảm ứng giữ nút LÊN / XUỐNG
local function setupTouchEvents(button, callback)
	button.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			callback(true)
		end
	end)
	button.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			callback(false)
		end
	end)
end

setupTouchEvents(up, function(state) isUpBtnDown = state end)
setupTouchEvents(down, function(state) isDownBtnDown = state end)

-- HÀM BẬT / TẮT BAY
local function toggleFly()
	local char = LocalPlayer.Character
	if not char then return end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum then return end

	nowe = not nowe

	if nowe then
		onof.Text = "TẮT"
		onof.BackgroundColor3 = Color3.fromRGB(0, 180, 0)
		hum.PlatformStand = true
		
		currentPos = hrp.Position + Vector3.new(0, 0.5, 0)

		for _, v in pairs(hum:GetPlayingAnimationTracks()) do
			v:Stop()
		end

		flyConnection = RunService.RenderStepped:Connect(function(deltaTime)
			hrp.AssemblyLinearVelocity = Vector3.zero
			hrp.AssemblyAngularVelocity = Vector3.zero

			local moveDir = hum.MoveDirection
			local speedMultiplier = speeds * 45 * deltaTime

			local nextPos = currentPos

			if moveDir.Magnitude > 0 then
				local cameraCFrame = Camera.CFrame
				local relativeMove = cameraCFrame:VectorToObjectSpace(moveDir)
				local flyVector = (cameraCFrame.LookVector * -relativeMove.Z) + (cameraCFrame.RightVector * relativeMove.X)
				nextPos = nextPos + (flyVector * speedMultiplier)
			end

			-- Nút LÊN / XUỐNG
			if isUpBtnDown then
				nextPos = nextPos + Vector3.new(0, speedMultiplier, 0)
			end
			if isDownBtnDown then
				nextPos = nextPos - Vector3.new(0, speedMultiplier, 0)
			end

			currentPos = checkCollision(currentPos, nextPos)

			local hoverY = math.sin(tick() * 3.5) * 0.35
			local hoverOffset = Vector3.new(0, hoverY, 0)

			local targetCFrame
			if moveDir.Magnitude > 0 then
				local cameraCFrame = Camera.CFrame
				local relativeMove = cameraCFrame:VectorToObjectSpace(moveDir)
				local flyVector = (cameraCFrame.LookVector * -relativeMove.Z) + (cameraCFrame.RightVector * relativeMove.X)

				targetCFrame = CFrame.new(currentPos + hoverOffset, currentPos + hoverOffset + flyVector) * CFrame.Angles(-math.rad(90), 0, 0)
			else
				local lookVec = Camera.CFrame.LookVector
				targetCFrame = CFrame.new(currentPos + hoverOffset, currentPos + hoverOffset + Vector3.new(lookVec.X, 0, lookVec.Z))
			end

			local lerpSpeed = math.clamp(deltaTime * 9, 0, 1)
			hrp.CFrame = hrp.CFrame:Lerp(targetCFrame, lerpSpeed)
		end)
	else
		stopFly()
	end
end

onof.MouseButton1Click:Connect(toggleFly)

	if _G.WuWFlyCleanup == stopFly then

end)

-- XỬ LÝ PHÍM TẮT MÁY TÍNH
local function updateKeybind()
	local text = string.upper(keyInput.Text)
	if text == "" then return end
	
	for _, key in ipairs(Enum.KeyCode:GetEnumItems()) do
		if key.Name == text or key.Name == "Number" .. text then
			selectedKey = key
			keyInput.Text = key.Name:gsub("Number", "")
			return
		end
	end
end

keyInput.FocusLost:Connect(updateKeybind)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.UserInputType == Enum.UserInputType.Keyboard then
		if selectedKey and input.KeyCode == selectedKey then
			toggleFly()
		end
	end
end)

-- Tự tắt khi chết/reset
LocalPlayer.CharacterAdded:Connect(function()
	stopFly()
end)
