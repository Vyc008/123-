-- ========================
-- HÀM LƯU CÀI ĐẶT — LƯU TOÀN BỘ
-- ========================
local function SaveSettings()
    if not writefile or not HttpService then
        return false, "Hàm lưu không khả dụng"
    end
    local success, err = pcall(function()
        local jsonData = HttpService:JSONEncode(mySettings)
        writefile(settingsFileName, jsonData)
    end)
    return success, err or "Đã lưu"
end

-- ========================
-- HÀM TẢI CÀI ĐẶT — TẢI ĐẦY ĐỦ
-- ========================
local function LoadSettings()
    -- Giữ nguyên mặc định nếu không có file
    local defaultSettings = {
        verticalPower = 50,
        keyBind = "F",
        tiltWithCamera = true,
        enableAnimation = true,
        enableHover = true
    }
    -- Bắt đầu với mặc định
    for k, v in pairs(defaultSettings) do
        mySettings[k] = v
    end
    -- Kiểm tra hàm đọc
    if not readfile or not isfile or not HttpService then
        return false, "Hàm đọc không khả dụng"
    end
    if not isfile(settingsFileName) then
        return true, "Chưa có file, dùng mặc định"
    end
    -- Đọc và ghi đè từng giá trị
    local success, err = pcall(function()
        local jsonData = readfile(settingsFileName)
        local loaded = HttpService:JSONDecode(jsonData)
        if loaded then
            -- Lực lên/xuống
            mySettings.verticalPower = tonumber(loaded.verticalPower) or 50
            -- Phím tắt
            mySettings.keyBind = string.upper(loaded.keyBind or "F")
            -- Xoay theo camera
            if loaded.tiltWithCamera ~= nil then mySettings.tiltWithCamera = loaded.tiltWithCamera end
            -- Nằm khi bay
            if loaded.enableAnimation ~= nil then mySettings.enableAnimation = loaded.enableAnimation end
            -- Nhấp nhô
            if loaded.enableHover ~= nil then mySettings.enableHover = loaded.enableHover end
        end
    end)
    return success, err or "Đã tải"
end

-- Tải cài đặt khi khởi động
LoadSettings()

local function setupTouch(btn, callback)
    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            callback(true)
        end
    end)
    btn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            callback(false)
        end
    end)
end

-- BẬT / TẮT BAY
-- ========================
local function toggleFly()
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    nowe = not nowe
    if nowe then
        onof.Text = "TẮT"
        onof.BackgroundColor3 = Color3.fromRGB(0, 180, 0)
        hum.PlatformStand = true
        currentPos = hrp.Position + Vector3.new(0, 3, 0)
        currentLieAngle = 0

        local rayParams = RaycastParams.new()
        rayParams.FilterDescendantsInstances = {char}
        rayParams.FilterType = Enum.RaycastFilterType.Exclude

        flyConnection = RunService.RenderStepped:Connect(function(dt)
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero

            local moveDir = hum.MoveDirection
            local isMoving = moveDir.Magnitude > 0.01 or isUpBtnDown or isDownBtnDown
            local spd = speeds * 45 * dt
            local vSpd = mySettings.verticalPower * dt * speeds
            local nextPos = currentPos

            -- Di chuyển ngang
            if moveDir.Magnitude > 0 then
                local camCF = Camera.CFrame
                local rel = camCF:VectorToObjectSpace(moveDir)
                local dir = (camCF.LookVector * -rel.Z) + (camCF.RightVector * rel.X)
                nextPos += dir * spd
            end
            -- Lên / Xuống
            if isUpBtnDown then nextPos += Vector3.new(0, vSpd, 0) end
            if isDownBtnDown then nextPos -= Vector3.new(0, vSpd, 0) end

            -- Kiểm tra tường + đất — SỬA LỖI DỊCH CHUYỂN
            if not noclip then
                local delta = nextPos - currentPos
                if delta.Magnitude > 0.001 then
                    local wallRay = workspace:Raycast(currentPos, delta.Unit * (delta.Magnitude + 2), rayParams)
                    if wallRay then
                        local normal = wallRay.Normal
                        local dot = delta:Dot(normal)
                        if dot < 0 then delta -= normal * dot end
                        nextPos = currentPos + delta
                    end
                end
                -- Giữ khoảng cách đất — GIỮ NGUYÊN X, Z
                local groundRay = workspace:Raycast(nextPos + Vector3.new(0, 5, 0), Vector3.new(0, -15, 0), rayParams)
                if groundRay then
                    local groundY = groundRay.Position.Y
                    local safe = 3.5
                    if nextPos.Y < groundY + safe then
                        nextPos = Vector3.new(nextPos.X, groundY + safe, nextPos.Z)
                    end
                end
            end

            currentPos = nextPos

            -- Hiệu ứng nhấp nhô
            local hoverY = mySettings.enableHover and math.sin(tick() * 3.5) * 0.35 or 0
            local hoverOff = Vector3.new(0, hoverY, 0)

            -- Nằm / Đứng mượt mà
            if mySettings.enableAnimation then
                targetLieAngle = isMoving and -90 or 0
            else
                targetLieAngle = 0
            end
            local diff = targetLieAngle - currentLieAngle
            currentLieAngle += diff * math.min(dt * 4, 1)

            -- Hướng nhìn
            local lookVec = Camera.CFrame.LookVector
            if not mySettings.tiltWithCamera then
                lookVec = Vector3.new(lookVec.X, 0, lookVec.Z).Unit
            end

            -- Áp dụng vị trí + góc
            local lookAt = currentPos + hoverOff + lookVec
            local targetCF = CFrame.new(currentPos + hoverOff, lookAt) 
                * CFrame.Angles(math.rad(currentLieAngle), 0, 0)
            hrp.CFrame = hrp.CFrame:Lerp(targetCF, math.clamp(dt * 9, 0, 1))
        end)
    else
        stopFly()
    end
end

-- CẬP NHẬT PHÍM TẮT
-- ========================
local function updateKeybind()
    local txt = string.upper(keyInput.Text)
    if txt == "" then return end
    for _, key in ipairs(Enum.KeyCode:GetEnumItems()) do
        if key.Name == txt or key.Name == "Number" .. txt then
            selectedKey = key
            keyInput.Text = key.Name:gsub("Number", "")
            mySettings.keyBind = keyInput.Text
            SaveSettings() -- LƯU NGAY KHI THAY ĐỔI PHÍM
            return
        end
    end
end
keyInput.FocusLost:Connect(updateKeybind)

-- BẬT/TẮT BAY BẰNG PHÍM
UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.UserInputType == Enum.UserInputType.Keyboard then
        if selectedKey and input.KeyCode == selectedKey then
            toggleFly()
        end
    end
end)
