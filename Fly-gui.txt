local settingsFile = "FlyGuiSettings_Ctgv.json"
local mySettings = {
	verticalPower = 50,
	keyBind = "F"
}

-- HÀM LƯU/TẢI CÀI ĐẶT
local function SaveSettings()
	if writefile then
		pcall(function()
			writefile(settingsFile, HttpService:JSONEncode(mySettings))
		end)
	end
end

local function LoadSettings()
	if readfile and isfile then
		pcall(function()
			if isfile(settingsFile) then
				local data = HttpService:JSONDecode(readfile(settingsFile))
				if data then
					mySettings.verticalPower = data.verticalPower or 50
					mySettings.keyBind = data.keyBind or "F"
				end
			end
		end)
	end
end
LoadSettings()

-- HÀM XỬ LÝ NOCLIP
local function setNoclip(state)
	noclip = state
	if noclip then
		noclipBtn.Text = "NC: ON"
		noclipBtn.BackgroundColor3 = Color3.fromRGB(0, 180, 0)
		if not noclipConnection then
			noclipConnection = RunService.Stepped:Connect(function()
				local char = LocalPlayer.Character
				if char then
					for _, part in ipairs(char:GetDescendants()) do
						if part:IsA("BasePart") then
							part.CanCollide = false
						end
					end
				end
			end)
		end
	else
		noclipBtn.Text = "NC: OFF"
		noclipBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
		if noclipConnection then
			noclipConnection:Disconnect()
			noclipConnection = nil
		end
	end
end

noclipBtn.MouseButton1Click:Connect(function()
	setNoclip(not noclip)
end)

-- HÀM TẮT BAY
local function stopFly()
	nowe = false
	onof.Text = "BAY"
	onof.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
	
	local char = LocalPlayer.Character
	if char then
		local hum = char:FindFirstChildOfClass("Humanoid")
		if hum then
			hum.PlatformStand = false
		end
	end
	
	if flyConnection then
		flyConnection:Disconnect()
		flyConnection = nil
	end
end

	if speeds > 1 then

local function setupTouchEvents(button, callback)
	button.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			callback(true)
		end
	end)
	button.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			callback(false)
		end
	end)
end

setupTouchEvents(up, function(state) isUpBtnDown = state end)
setupTouchEvents(down, function(state) isDownBtnDown = state end)

RunService.Heartbeat:Connect(function()
	if not nowe and (isUpBtnDown or isDownBtnDown) then
		local char = LocalPlayer.Character
		if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChildOfClass("Humanoid") then
			local hrp = char.HumanoidRootPart
			local jumpForce = mySettings.verticalPower 

			if isUpBtnDown then
				hrp.AssemblyLinearVelocity = Vector3.new(hrp.AssemblyLinearVelocity.X, jumpForce, hrp.AssemblyLinearVelocity.Z)
			elseif isDownBtnDown then
				hrp.AssemblyLinearVelocity = Vector3.new(hrp.AssemblyLinearVelocity.X, -jumpForce, hrp.AssemblyLinearVelocity.Z)
			end
		end
	end
end)

-- HÀM BẬT / TẮT BAY KÈM HỆ THỐNG TRƯỢT TƯỜNG & CÁCH ĐẤT
local function toggleFly()
	local char = LocalPlayer.Character
	if not char then return end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum then return end

	nowe = not nowe

	if nowe then
		onof.Text = "TẮT"
		onof.BackgroundColor3 = Color3.fromRGB(0, 180, 0)
		hum.PlatformStand = true
		
		currentPos = hrp.Position + Vector3.new(0, 3, 0)

		for _, v in pairs(hum:GetPlayingAnimationTracks()) do
			v:Stop()
		end

		local raycastParams = RaycastParams.new()
		raycastParams.FilterDescendantsInstances = {char}
		raycastParams.FilterType = Enum.RaycastFilterType.Exclude

		flyConnection = RunService.RenderStepped:Connect(function(deltaTime)
			hrp.AssemblyLinearVelocity = Vector3.zero
			hrp.AssemblyAngularVelocity = Vector3.zero

			local moveDir = hum.MoveDirection
			local speedMultiplier = speeds * 45 * deltaTime
			local vertSpeedMultiplier = mySettings.verticalPower * deltaTime * speeds

			local nextPos = currentPos

			if moveDir.Magnitude > 0 then
				local cameraCFrame = Camera.CFrame
				local relativeMove = cameraCFrame:VectorToObjectSpace(moveDir)
				local flyVector = (cameraCFrame.LookVector * -relativeMove.Z) + (cameraCFrame.RightVector * relativeMove.X)
				nextPos = nextPos + (flyVector * speedMultiplier)
			end

			if isUpBtnDown then
				nextPos = nextPos + Vector3.new(0, vertSpeedMultiplier, 0)
			end
			if isDownBtnDown then
				nextPos = nextPos - Vector3.new(0, vertSpeedMultiplier, 0)
			end

			-- TÍNH TOÁN KHI NOCLIP TẮT (Trượt tường + Cách đất 1 stud)
			if not noclip then
				local moveDelta = nextPos - currentPos
				if moveDelta.Magnitude > 0 then
					local wallRay = workspace:Raycast(currentPos, moveDelta.Unit * (moveDelta.Magnitude + 2), raycastParams)
					if wallRay then
						local normal = wallRay.Normal
						local dot = moveDelta:Dot(normal)
						if dot < 0 then
							moveDelta = moveDelta - (normal * dot)
						end
						nextPos = currentPos + moveDelta
					end
				end

				local rayResult = workspace:Raycast(nextPos, Vector3.new(0, -10, 0), raycastParams)
				if rayResult then
					local groundY = rayResult.Position.Y
					local safeGroundDistance = 4 
					if nextPos.Y < groundY + safeGroundDistance then
						nextPos = Vector3.new(nextPos.X, groundY + safeGroundDistance, nextPos.Z)
					end
				end
			end

			currentPos = nextPos

			local hoverY = math.sin(tick() * 3.5) * 0.35
			local hoverOffset = Vector3.new(0, hoverY, 0)

			local targetCFrame
			if moveDir.Magnitude > 0 then
				local cameraCFrame = Camera.CFrame
				local relativeMove = cameraCFrame:VectorToObjectSpace(moveDir)
				local flyVector = (cameraCFrame.LookVector * -relativeMove.Z) + (cameraCFrame.RightVector * relativeMove.X)

				targetCFrame = CFrame.new(currentPos + hoverOffset, currentPos + hoverOffset + flyVector) * CFrame.Angles(-math.rad(90), 0, 0)
			else
				local lookVec = Camera.CFrame.LookVector
				targetCFrame = CFrame.new(currentPos + hoverOffset, currentPos + hoverOffset + Vector3.new(lookVec.X, 0, lookVec.Z))
			end

			local lerpSpeed = math.clamp(deltaTime * 9, 0, 1)
			hrp.CFrame = hrp.CFrame:Lerp(targetCFrame, lerpSpeed)
		end)
	else
		stopFly()
	end
end

onof.MouseButton1Click:Connect(toggleFly)

	stopFly()
	if _G.WuWFlyCleanup == stopFly then
		_G.WuWFlyCleanup = nil

local function updateKeybind()
	local text = string.upper(keyInput.Text)
	if text == "" then return end
	
	for _, key in ipairs(Enum.KeyCode:GetEnumItems()) do
		if key.Name == text or key.Name == "Number" .. text then
			selectedKey = key
			keyInput.Text = key.Name:gsub("Number", "")
			
			mySettings.keyBind = keyInput.Text
			SaveSettings()
			return
		end
	end
end

keyInput.FocusLost:Connect(updateKeybind)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.UserInputType == Enum.UserInputType.Keyboard then
		if selectedKey and input.KeyCode == selectedKey then
			toggleFly()
		end
	end
end)

LocalPlayer.CharacterAdded:Connect(function()
	stopFly()
end)
