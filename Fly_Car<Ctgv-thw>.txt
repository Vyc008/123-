local function ApplyMinimalStyle(element, isCircle)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = isCircle and UDim.new(1, 0) or UDim.new(0, 6)
    corner.Parent = element

    local stroke = Instance.new("UIStroke")
    stroke.Color = Theme.Border
    stroke.Thickness = 1
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Parent = element
end

-- Hàm tạo Nút
local function CreateButton(btn, parent, text, size, pos, textSize, isCircle)
    btn.Parent = parent
    btn.BackgroundColor3 = Theme.Panel
    btn.Size = size
    if pos then btn.Position = pos end
    btn.Font = MainFont
    btn.Text = text
    btn.TextColor3 = Theme.Text
    btn.TextSize = textSize or 12
    if btn:IsA("TextButton") then btn.AutoButtonColor = true end
    ApplyMinimalStyle(btn, isCircle)
end

-- ================= HỆ THỐNG KÉO THẢ CỦA EDIT MODE =================
local function makeDraggable(guiObject)
    local dragging, dragInput, dragStart, startPos
    
    guiObject.InputBegan:Connect(function(input)
        if not isEditMode then return end 
        
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = guiObject.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    
    guiObject.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            guiObject.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

-- ================= HỆ THỐNG LƯU VÀ TẢI CẤU HÌNH =================
local function SaveConfig()
    if type(writefile) ~= "function" then
        SaveButton.Text = "NOT SUPPORTED"
        task.wait(1.5)
        SaveButton.Text = "LƯU CẤU HÌNH"
        return
    end

    local configData = {
        Speed = SpeedTextBox.Text,
        FwF = flightWithoutFlyEnabled,
        FloatingFlyVisible = isFloatingFlyVisible,
        FloatingCtrlVisible = isFloatingControllerVisible,
        Positions = {
            MainFrame = {MainFrame.Position.X.Scale, MainFrame.Position.X.Offset, MainFrame.Position.Y.Scale, MainFrame.Position.Y.Offset},
            FloatingFly = {FloatingFlyButton.Position.X.Scale, FloatingFlyButton.Position.X.Offset, FloatingFlyButton.Position.Y.Scale, FloatingFlyButton.Position.Y.Offset},
            FloatingCtrl = {FloatingControllerButton.Position.X.Scale, FloatingControllerButton.Position.X.Offset, FloatingControllerButton.Position.Y.Scale, FloatingControllerButton.Position.Y.Offset},
            CtrlFrame = {ControllerFrame.Position.X.Scale, ControllerFrame.Position.X.Offset, ControllerFrame.Position.Y.Scale, ControllerFrame.Position.Y.Offset},
            LeftSide = {LeftSideFrame.Position.X.Scale, LeftSideFrame.Position.X.Offset, LeftSideFrame.Position.Y.Scale, LeftSideFrame.Position.Y.Offset}
        }
    }

    if not isfolder(ConfigFolder) then makefolder(ConfigFolder) end
    writefile(ConfigFile, HttpService:JSONEncode(configData))

    SaveButton.Text = "ĐÃ LƯU ✓"
    task.wait(1.5)
    SaveButton.Text = "LƯU CẤU HÌNH"
end

local function LoadConfig()
    if type(readfile) == "function" and isfile(ConfigFile) then
        local success, configData = pcall(function() return HttpService:JSONDecode(readfile(ConfigFile)) end)
        
        if success and configData then
            if configData.Speed then SpeedTextBox.Text = tostring(configData.Speed) end
            
            if configData.FwF ~= nil then
                flightWithoutFlyEnabled = configData.FwF
                FwFButton.Text = flightWithoutFlyEnabled and "FwF: ON" or "FwF: OFF"
                FwFButton.BackgroundColor3 = flightWithoutFlyEnabled and Theme.Active or Theme.Panel
                FwFButton.TextColor3 = flightWithoutFlyEnabled and Theme.Background or Theme.Text
            end

            if configData.FloatingFlyVisible ~= nil then
                isFloatingFlyVisible = configData.FloatingFlyVisible
                FloatingFlyButton.Visible = isFloatingFlyVisible
                ToggleFloatingButton.BackgroundColor3 = isFloatingFlyVisible and Theme.Active or Theme.Panel
                ToggleFloatingButton.TextColor3 = isFloatingFlyVisible and Theme.Background or Theme.Text
            end

            if configData.FloatingCtrlVisible ~= nil then
                isFloatingControllerVisible = configData.FloatingCtrlVisible
                FloatingControllerButton.Visible = isFloatingControllerVisible
                ToggleFloatingControllerButton.BackgroundColor3 = isFloatingControllerVisible and Theme.Active or Theme.Panel
                ToggleFloatingControllerButton.TextColor3 = isFloatingControllerVisible and Theme.Background or Theme.Text
            end

            if configData.Positions then
                local p = configData.Positions
                if p.MainFrame then MainFrame.Position = UDim2.new(unpack(p.MainFrame)) end
                if p.FloatingFly then FloatingFlyButton.Position = UDim2.new(unpack(p.FloatingFly)) end
                if p.FloatingCtrl then FloatingControllerButton.Position = UDim2.new(unpack(p.FloatingCtrl)) end
                if p.CtrlFrame then ControllerFrame.Position = UDim2.new(unpack(p.CtrlFrame)) end
                if p.LeftSide then LeftSideFrame.Position = UDim2.new(unpack(p.LeftSide)) end
            end
        end
    end
end

-- ================= LOGIC VẬT LÝ =================
local function setupFlyInstances(character)
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    if character.HumanoidRootPart:FindFirstChild(velocityHandlerName) then character.HumanoidRootPart[velocityHandlerName]:Destroy() end
    if character.HumanoidRootPart:FindFirstChild(gyroHandlerName) then character.HumanoidRootPart[gyroHandlerName]:Destroy() end
    
    VelocityHandler = Instance.new("BodyVelocity")
    VelocityHandler.Name = velocityHandlerName
    VelocityHandler.Parent = character.HumanoidRootPart
    VelocityHandler.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    VelocityHandler.Velocity = Vector3.new()
    
    GyroHandler = Instance.new("BodyGyro")
    GyroHandler.Name = gyroHandlerName
    GyroHandler.Parent = character.HumanoidRootPart
    GyroHandler.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    GyroHandler.P = 1000
    GyroHandler.D = 50
    GyroHandler.CFrame = character.HumanoidRootPart.CFrame
end

local function EnableFlight()
    if player.Character then setupFlyInstances(player.Character) end
end

local function DisableFlight()
    if VelocityHandler then VelocityHandler:Destroy() VelocityHandler = nil end
    if GyroHandler then GyroHandler:Destroy() GyroHandler = nil end
end

local function UpdateJoystickPosition(inputPosition)
    local center = JoystickBG.AbsolutePosition + (JoystickBG.AbsoluteSize / 2)
    local delta = Vector2.new(inputPosition.X, inputPosition.Y) - center
    local distance = delta.Magnitude

    if distance > joystickMaxRadius then
        delta = delta.Unit * joystickMaxRadius
    end

    JoystickKnob.Position = UDim2.new(0.5, delta.X - 30, 0.5, delta.Y - 30)
    joystickVector = Vector2.new(delta.X / joystickMaxRadius, delta.Y / joystickMaxRadius)
end

local function setupButtonConnections()
    local function joystickDragStart(input)
        if isEditMode then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if not isDraggingJoystick then
                isDraggingJoystick = true
                activeDragInput = input
                UpdateJoystickPosition(input.Position)
            end
        end
    end

    table.insert(activeConnections, JoystickBG.InputBegan:Connect(joystickDragStart))
    table.insert(activeConnections, JoystickKnob.InputBegan:Connect(joystickDragStart))

    table.insert(activeConnections, UserInputService.InputChanged:Connect(function(input)
        if isDraggingJoystick and input == activeDragInput then
            UpdateJoystickPosition(input.Position)
        end
    end))

    table.insert(activeConnections, UserInputService.InputEnded:Connect(function(input)
        if isDraggingJoystick and input == activeDragInput then
            isDraggingJoystick = false
            activeDragInput = nil 
            JoystickKnob.Position = UDim2.new(0.5, -30, 0.5, -30)
            joystickVector = Vector2.new(0, 0)
        end
    end))

    table.insert(activeConnections, IncreaseButton.MouseButton1Click:Connect(function()
        local currentSpeed = tonumber(SpeedTextBox.Text) or 1
        SpeedTextBox.Text = tostring(math.round((currentSpeed + 0.1) * 10) / 10)
    end))
    table.insert(activeConnections, DecreaseButton.MouseButton1Click:Connect(function()
        local currentSpeed = tonumber(SpeedTextBox.Text) or 1
        SpeedTextBox.Text = tostring(math.max(0.1, math.round((currentSpeed - 0.1) * 10) / 10))
    end))
    
    local rightButtons = {
        {button = UpButton, dir = "up"}, {button = DownButton, dir = "down"},
        {button = RotateLeftButton, dir = "rotateLeft"}, {button = RotateRightButton, dir = "rotateRight"}
    }
    for _, config in ipairs(rightButtons) do
        table.insert(activeConnections, config.button.MouseButton1Down:Connect(function() 
            if isEditMode then return end
            table.insert(movementDirection, config.dir) 
        end))
        table.insert(activeConnections, config.button.MouseButton1Up:Connect(function()
            local index = table.find(movementDirection, config.dir)
            while index do table.remove(movementDirection, index) index = table.find(movementDirection, config.dir) end
        end))
    end
    
    table.insert(activeConnections, AntiLockButton.MouseButton1Click:Connect(function()
        isAntiLockOn = not isAntiLockOn
        AntiLockButton.Text = isAntiLockOn and "A-L: ON" or "A-L: OFF"
        AntiLockButton.BackgroundColor3 = isAntiLockOn and Theme.Active or Theme.Panel
        AntiLockButton.TextColor3 = isAntiLockOn and Theme.Background or Theme.Text
        if isAntiLockOn and isPitchOn then
            isPitchOn = false
            PitchButton.Text = "PITCH: OFF"
            PitchButton.BackgroundColor3 = Theme.Panel
            PitchButton.TextColor3 = Theme.Text
        end
    end))
    
    table.insert(activeConnections, PitchButton.MouseButton1Click:Connect(function()
        isPitchOn = isAntiLockOn and false or not isPitchOn
        PitchButton.Text = isPitchOn and "PITCH: ON" or "PITCH: OFF"
        PitchButton.BackgroundColor3 = isPitchOn and Theme.Active or Theme.Panel
        PitchButton.TextColor3 = isPitchOn and Theme.Background or Theme.Text
    end))
end

local function handleInput()
    local character = player.Character
    if not character or not character:FindFirstChild("Humanoid") then return end
    local humanoid = character.Humanoid
    
    if FlyEnabled and (humanoid.SeatPart or flightWithoutFlyEnabled) then
        local rootPart = character:FindFirstChild("HumanoidRootPart")
        if not rootPart or not VelocityHandler then return end
        
        local gyro = rootPart:FindFirstChild(gyroHandlerName)
        local speed = tonumber(SpeedTextBox.Text) or 1
        local velocity = Vector3.new()
        
        local movementCFrame = (isPitchOn and gyro) and gyro.CFrame or (gyro and CFrame.new(gyro.Parent.Position, gyro.Parent.Position + Vector3.new(gyro.CFrame.LookVector.X, 0, gyro.CFrame.LookVector.Z)) or rootPart.CFrame)
        
        local joyForward = -joystickVector.Y * forwardVelocity
        local joyRight = joystickVector.X * rightVelocity
        
        velocity += movementCFrame.LookVector * joyForward
        velocity += movementCFrame.RightVector * joyRight

        for _, direction in ipairs(movementDirection) do
            if direction == "up" then velocity += Vector3.new(0, upVelocity, 0)
            elseif direction == "down" then velocity += Vector3.new(0, -downVelocity, 0)
            elseif direction == "rotateLeft" and gyro then gyro.CFrame *= CFrame.Angles(0, math.rad(-rotationSpeed), 0)
            elseif direction == "rotateRight" and gyro then gyro.CFrame *= CFrame.Angles(0, math.rad(rotationSpeed), 0)
            end
        end
        
        local thumbstickDirection = Vector2.new(0, 0)
        pcall(function() thumbstickDirection = require(player.PlayerScripts:WaitForChild("PlayerModule"):WaitForChild("ControlModule")):GetMoveVector() end)
        
        if joystickVector.Magnitude > 0 or #movementDirection > 0 or thumbstickDirection.Magnitude > 0 then
            local combinedVelocity = velocity + (movementCFrame.RightVector * thumbstickDirection.X * forwardVelocity * 2 + movementCFrame.LookVector * -thumbstickDirection.Z * forwardVelocity * 2)
            
            if isAntiLockOn and humanoid.SeatPart then
                local sCF = humanoid.SeatPart.CFrame
                local seatVelocity = Vector3.new()
                
                seatVelocity += sCF.LookVector * joyForward
                seatVelocity += sCF.RightVector * joyRight

                for _, direction in ipairs(movementDirection) do
                    if direction == "up" then seatVelocity += sCF.UpVector * upVelocity
                    elseif direction == "down" then seatVelocity += -sCF.UpVector * downVelocity
                    end
                end
                VelocityHandler.Velocity = seatVelocity * speed
            else
                VelocityHandler.Velocity = combinedVelocity * speed
            end
        else
            VelocityHandler.Velocity = Vector3.new()
        end
        
        if not isAntiLockOn then
            local camera = workspace.CurrentCamera
            if isPitchOn and gyro then gyro.CFrame = camera.CFrame
            elseif gyro then gyro.CFrame = CFrame.new(gyro.Parent.Position, gyro.Parent.Position + (camera.CFrame.LookVector * Vector3.new(1, 0, 1)))
            end
        end
    else
        if VelocityHandler then VelocityHandler.Velocity = Vector3.new() end
    end
end

local function toggleFlyState()
    FlyEnabled = not FlyEnabled
    local humanoid = player.Character and player.Character:FindFirstChild("Humanoid")
    local text = FlyEnabled and "FLY: ON" or "FLY: OFF"
    local bgColor = FlyEnabled and Theme.Active or Theme.Panel
    local txtColor = FlyEnabled and Theme.Background or Theme.Text
    
    FlyButton.Text, FlyButton.BackgroundColor3, FlyButton.TextColor3 = text, bgColor, txtColor
    
    if not isEditMode then
        FloatingFlyButton.Text = text
        FloatingFlyButton.BackgroundColor3 = bgColor
        FloatingFlyButton.TextColor3 = txtColor
    end
    
    if FlyEnabled then
        if flightWithoutFlyEnabled then EnableFlight()
        elseif humanoid then
            if seatConnection then seatConnection:Disconnect() end
            if unseatConnection then unseatConnection:Disconnect() end
            if humanoid.SeatPart then EnableFlight()
            else seatConnection = humanoid:GetPropertyChangedSignal("SeatPart"):Connect(function() if humanoid.SeatPart then EnableFlight() end end) end
            unseatConnection = humanoid:GetPropertyChangedSignal("SeatPart"):Connect(function() if not humanoid.SeatPart then DisableFlight() end end)
        end
    else
        DisableFlight()
        if seatConnection then seatConnection:Disconnect() end
        if unseatConnection then unseatConnection:Disconnect() end
    end
end

local function toggleControllerState()
    controllerEnabled = not controllerEnabled
    ControllerGui.Enabled = controllerEnabled
    LeftSideGui.Enabled = controllerEnabled
    
    local menuText = controllerEnabled and "Hide Controller" or "Show Controller"
    local floatText = controllerEnabled and "CTRL: ON" or "CTRL: OFF"
    local bgColor = controllerEnabled and Theme.Active or Theme.Panel
    local txtColor = controllerEnabled and Theme.Background or Theme.Text
    
    ControllerButton.Text, ControllerButton.BackgroundColor3, ControllerButton.TextColor3 = menuText, bgColor, txtColor
    
    if not isEditMode then
        FloatingControllerButton.Text = floatText
        FloatingControllerButton.BackgroundColor3 = bgColor
        FloatingControllerButton.TextColor3 = txtColor
    end
end

local function toggleEditMode()
    isEditMode = not isEditMode
    
    EditUIButton.Text = isEditMode and "EDIT UI: ON" or "EDIT UI (MOVE/PIN)"
    EditUIButton.BackgroundColor3 = isEditMode and Color3.fromRGB(255, 170, 0) or Theme.Panel
    EditUIButton.TextColor3 = isEditMode and Theme.Background or Theme.Text
    
    local frameBgTrans = isEditMode and 0.4 or 1
    local frameBgColor = isEditMode and Color3.fromRGB(255, 170, 0) or Theme.Background
    
    ControllerFrame.BackgroundTransparency = frameBgTrans
    ControllerFrame.BackgroundColor3 = frameBgColor
    LeftSideFrame.BackgroundTransparency = frameBgTrans
    LeftSideFrame.BackgroundColor3 = frameBgColor
    
    if isEditMode then
        FloatingFlyButton.BackgroundColor3 = Color3.fromRGB(255, 170, 0)
        FloatingFlyButton.TextColor3 = Theme.Background
        FloatingControllerButton.BackgroundColor3 = Color3.fromRGB(255, 170, 0)
        FloatingControllerButton.TextColor3 = Theme.Background
    else
        FloatingFlyButton.BackgroundColor3 = FlyEnabled and Theme.Active or Theme.Panel
        FloatingFlyButton.TextColor3 = FlyEnabled and Theme.Background or Theme.Text
        FloatingControllerButton.BackgroundColor3 = controllerEnabled and Theme.Active or Theme.Panel
        FloatingControllerButton.TextColor3 = controllerEnabled and Theme.Background or Theme.Text
    end
end
