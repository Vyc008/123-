-- LOGIC XỬ LÝ TRẠNG THÁI VÀ GIAO DIỆN
-- ==========================================

local isGlitchEnabled = false
local isFloatingVisible = false
local hasJumped = false
local stateConn = nil

-- Hàm cập nhật đồng bộ trạng thái bật/tắt cho cả 2 nút
local function updateGlitchState(state)
    isGlitchEnabled = state
    if isGlitchEnabled then
        ToggleButton.BackgroundColor3 = Color3.fromRGB(40, 180, 80)
        ToggleButton.Text = "Trạng thái: ON"
        FloatingButton.BackgroundColor3 = Color3.fromRGB(40, 180, 80)
        FloatingButton.Text = "Glitch: ON"
    else
        ToggleButton.BackgroundColor3 = Color3.fromRGB(180, 40, 40)
        ToggleButton.Text = "Trạng thái: OFF"
        FloatingButton.BackgroundColor3 = Color3.fromRGB(180, 40, 40)
        FloatingButton.Text = "Glitch: OFF"
    end
end

-- Nhấn nút bật/tắt chính trong menu
ToggleButton.MouseButton1Click:Connect(function()
    updateGlitchState(not isGlitchEnabled)
end)

-- Nhấn nút nổi bên ngoài để bật/tắt
FloatingButton.MouseButton1Click:Connect(function()
    updateGlitchState(not isGlitchEnabled)
end)

-- Nhấn nút ẩn/hiện nút nổi (4 phần)
FloatToggleMenuBtn.MouseButton1Click:Connect(function()
    isFloatingVisible = not isFloatingVisible
    FloatingButton.Visible = isFloatingVisible
    if isFloatingVisible then
        FloatToggleMenuBtn.BackgroundColor3 = Color3.fromRGB(40, 120, 180)
        FloatToggleMenuBtn.Text = "Nút nổi: Hiện"
    else
        FloatToggleMenuBtn.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
        FloatToggleMenuBtn.Text = "Nút nổi: Ẩn"
    end
end)

-- Hàm theo dõi trạng thái nhảy thực tế của nhân vật
local function setupCharacter(char)
    if stateConn then stateConn:Disconnect() end
    hasJumped = false
    
    local humanoid = char:WaitForChild("Humanoid", 5)
    if humanoid then
        stateConn = humanoid.StateChanged:Connect(function(oldState, newState)
            if newState == Enum.HumanoidStateType.Jumping then
                hasJumped = true
            elseif newState == Enum.HumanoidStateType.Running or newState == Enum.HumanoidStateType.Landed then
                hasJumped = false
            end
        end)
    end
end

LocalPlayer.CharacterAdded:Connect(setupCharacter)
if LocalPlayer.Character then
    setupCharacter(LocalPlayer.Character)
end

-- ==========================================
-- VÒNG LẶP XỬ LÝ VẬT LÝ & CHECK VẬT CẢN
-- ==========================================
RunService.RenderStepped:Connect(function(dt)
    if not isGlitchEnabled then return end
    
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    
    if not hrp or not humanoid then return end

    local customSpeed = tonumber(SpeedBox.Text) or 36
    local isInAir = (humanoid.FloorMaterial == Enum.Material.Air)
    local currentVelocity = hrp.AssemblyLinearVelocity

    if isInAir and humanoid.MoveDirection.Magnitude > 0 and hasJumped then
        local moveDir = humanoid.MoveDirection.Unit
        
        -- Tạo tia Raycast dò đường phía trước
        local rayOrigin = hrp.Position
        local rayDistance = 3.5
        local rayDirection = moveDir * rayDistance
        
        local rayParams = RaycastParams.new()
        rayParams.FilterDescendantsInstances = {char}
        rayParams.FilterType = Enum.RaycastFilterType.Exclude
        
        local rayHit = Workspace:Raycast(rayOrigin, rayDirection, rayParams)
        
        -- Nếu gặp vật cản thì dừng lao lại
        if rayHit and rayHit.Instance and rayHit.Instance.CanCollide then
            hasJumped = false 
        else
            hrp.AssemblyLinearVelocity = Vector3.new(moveDir.X * customSpeed, currentVelocity.Y, moveDir.Z * customSpeed)
        end
    end
end)
